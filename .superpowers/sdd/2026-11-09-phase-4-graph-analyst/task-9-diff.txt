+function extractFactValue(fact: Record<string, unknown> | undefined): string {
+  if (!fact) return "—"
+  if (typeof fact.value === "string" && fact.value) return fact.value
+  if (typeof fact.object_text === "string" && fact.object_text)
+    return fact.object_text
+  if (typeof fact.val === "string" && fact.val) return fact.val
+  if (typeof fact.text === "string" && fact.text) return fact.text
+  return Object.keys(fact).length > 0 ? JSON.stringify(fact) : "—"
+}

+function extractFactSource(fact: Record<string, unknown> | undefined): string {
+  if (!fact) return "Unknown source"
+  if (typeof fact.documentTitle === "string" && fact.documentTitle)
+    return fact.documentTitle
+  if (typeof fact.sourceName === "string" && fact.sourceName)
+    return fact.sourceName
+  if (typeof fact.sourceTitle === "string" && fact.sourceTitle)
+    return fact.sourceTitle
+  if (typeof fact.document_name === "string" && fact.document_name)
+    return fact.document_name
+  if (typeof fact.source === "string" && fact.source) return fact.source
+  if (typeof fact.extractor === "string" && fact.extractor)
+    return fact.extractor
+  return "Unknown source"

+export async function runDriftAnalysisAction(
+  ontologyIdOrInput: string | { ontologyId: string },
+  dbInstance?: DbOrTx
+): Promise<RunDriftAnalysisResult> {
+  const ontologyId =
+    typeof ontologyIdOrInput === "string"
+      ? ontologyIdOrInput
+      : ontologyIdOrInput?.ontologyId

+    aggregateType: "ontology",
+    aggregateId: ontologyId.trim(),
+    payload: {
+      ontologyId: ontologyId.trim(),
+      groupKey,
+    },
+  })
+
+  return {
+    success: true,
+    eventId,
+    jobId,
+  }

+  const insertedEvents: Array<Record<string, unknown>> = []
+  const insertedJobs: Array<Record<string, unknown>> = []
+
+  const mockDb = {
+    insert: () => ({
+      values: (val: Record<string, unknown>) => {
+        if ("event_type" in val) {
+          insertedEvents.push(val)
+          return {
+            returning: () => Promise.resolve([{ id: "mock-evt-drift-123" }]),
+          }
+        }
+        if ("target_worker" in val) {
+          insertedJobs.push(val)
+          return {
+            returning: () => Promise.resolve([{ id: "mock-job-drift-456" }]),
+          }
+        }
+        return {
+          returning: () => Promise.resolve([{ id: "mock-id-789" }]),
+        }
+      },
+    }),
+  }
+
+  const result = await runDriftAnalysisAction(
+    "ont-drift-100",
+    mockDb as unknown as Parameters<typeof runDriftAnalysisAction>[1]
+  )
+
+  assert.equal(result.success, true)
+  assert.equal(result.eventId, "mock-evt-drift-123")
+  assert.equal(result.jobId, "mock-job-drift-456")
+
+  assert.equal(insertedEvents.length, 1)
+  assert.equal(insertedEvents[0].event_type, "analyst.drift_analysis_requested")
+  assert.equal(insertedEvents[0].aggregate_type, "ontology")
+  assert.equal(insertedEvents[0].aggregate_id, "ont-drift-100")
+
+  const payload = insertedEvents[0].payload as Record<string, unknown>
+  assert.equal(payload.ontologyId, "ont-drift-100")
+  assert.equal(payload.groupKey, "shared")
+
+  assert.equal(insertedJobs.length, 1)
+  assert.equal(insertedJobs[0].target_worker, "analyst")
+})
